# Token definitions for lexer.
# A token is defined by a class name followed by whitespace(s)
# and finally by the regex that defines it.
# Regex fragments can be used as helpers and aren't considered tokens.
# They are defined by prefixing their name with !.

# Fragment definitions.
!word_char      \w
!digit          \d

# Patterns to ignore
?WHITESPACE     [^\S\n\r]+
?NEWLINE        \r?\n

# Token definitions
SEMICOLON       ;
LEFT_PAREN      [(]
RIGHT_PAREN     [)]
LEFT_CURLY      [{]
RIGHT_CURLY     [}]
INTEGER         \d+
FLOAT           \d+[.]\d+
ASSIGNMENT      =
SUM             [+]
SUBTRACTION     -
MULTIPLICATION  [*]
DIVISION        /
EXPO            ^
EQUAL           ==
GREATER_THAN    >
LESS_THAN       <
GREATER_EQ_THAN >=
LESS_EQ_THAN    <=
COMMA           ,
DTYPE_INT       int
DTYPE_FLOAT     float
IDENTIFIER      [a-zA-Z]{word_char}*
